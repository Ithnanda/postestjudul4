Kunci Paket Soal 1: Perpustakaan
# 1
mkdir posttest_perpus && cd posttest_perpus
# 2 (Bebas 5 buku format Judul:Penulis:Tahun)
echo "Judul1:PenulisA:2020" > katalog.txt
echo "Judul2:PenulisA:2021" >> katalog.txt
echo "Judul3:PenulisB:2019" >> katalog.txt
echo "Judul4:PenulisC:2022" >> katalog.txt
echo "Judul5:PenulisD:2023" >> katalog.txt
# 3
sort katalog.txt > katalog_urut.txt
# 4 (Filter salah satu penulis di katalog.txt)
grep "PenulisA" katalog.txt > buku_penulis.txt
# 5
wc -l < katalog.txt > total_buku.txt
# 6 (nano pinjam.sh)
#!/bin/bash
read -p "Judul buku yang dipinjam: " judul
if grep -q "$judul" katalog.txt
then
    echo "$judul" >> pinjam.txt
    echo "Peminjaman dicatat"
else
    echo "Buku tidak ditemukan"
fi
# 7
chmod +x pinjam.sh
./pinjam.sh   # 2x judul ada, 1x judul tidak ada
# 8
wc -l < pinjam.txt > jml_pinjam.txt
# 9 (nano sisa.sh)
#!/bin/bash
total=$(cat total_buku.txt)
pinjam=$(cat jml_pinjam.txt)
sisa=$((total - pinjam))
echo "Buku tersedia: $sisa"
if [ $sisa -le 3 ]
then
    echo "Stok menipis"
fi
# 10
chmod +x sisa.sh
./sisa.sh | tee laporan_perpus.txt

# 8 (nano cek.sh)
#!/bin/bash
jumlah=$(cat jumlah.txt)
hadir=$(wc -l < hadir.txt)
if [ $hadir -ge $jumlah ]
then
    echo "Absensi lengkap"
else
    echo "Belum lengkap, kurang $((jumlah - hadir)) orang"
fi
# 9
chmod +x cek.sh
./cek.sh > status.txt
# 10
ls -l > daftar_file.txt && cat status.txt
Kunci Paket Soal 3: Kalkulasi Nilai
# 1
mkdir posttest_nilai && cd posttest_nilai
# 2-5 (nano nilai.sh)
#!/bin/bash
read -p "Nama: " nama
read -p "Nilai Laporan: " laporan
read -p "Nilai Posttest: " posttest
read -p "Nilai Praktik: " praktik

nilai=$((laporan * 30 / 100 + posttest * 30 / 100 + praktik * 40 / 100))
echo "Nilai akhir $nama: $nilai"

if [ $nilai -ge 85 ]
then
    grade="A"
elif [ $nilai -ge 70 ]
then
    grade="B"
else
    grade="C"
fi
echo "Grade: $grade"
echo "$nama:$nilai:$grade" >> rekap.txt
# 6
chmod +x nilai.sh
./nilai.sh   # (3x)
# 7
sort rekap.txt > rekap_urut.txt
# 8
grep ":A$" rekap.txt > lulus_A.txt
# 9
wc -l < rekap.txt > total.txt
# 10
cat rekap_urut.txt | tee laporan_akhir.txt
